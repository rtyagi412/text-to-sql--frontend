SELECT
    a.AccountNumber  AS [Account Number],
    c.FullName       AS [Customer Name],
    b.BranchName     AS [Branch],
    a.LedgerBalance  AS [Balance],
    a.LastTxnDate    AS [Last Transaction Date]
FROM dbo.Accounts a
JOIN dbo.Customers c ON c.CustomerId = a.CustomerId
JOIN dbo.Branches  b ON b.BranchId   = a.BranchId
WHERE a.AccountType = 'SAVINGS'
  AND b.Status      = 'ACTIVE'
  AND a.LastTxnDate < DATEADD(DAY, -180, GETDATE())
  AND a.Status      <> 'CLOSED'
ORDER BY b.BranchName, a.AccountNumber;

SELECT
    a.AccountNumber AS [Account Number],
    c.FullName AS [Customer Name],
    b.BranchName AS [Branch],
    a.LedgerBalance AS [Balance],
    a.LastTxnDate AS [Last Transaction Date]
FROM dbo.Accounts AS a
    INNER JOIN dbo.Customers AS c ON c.CustomerId = a.CustomerId
    INNER JOIN dbo.Branches AS b ON b.BranchId = a.BranchId
WHERE a.AccountType = 'SAVINGS'
    AND b.Status = 'ACTIVE'
    AND a.LastTxnDate < DATEADD(DAY, -180, GETDATE())
    AND a.Status <> 'CLOSED'
ORDER BY b.BranchName, a.AccountNumber;
